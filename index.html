  <script>

    const MAX_ALTITUDE = 500;

    const DIFFICULTY = {

      easy: { numBuildings: 12, baseSpeed: 1.0, maxSpeed: 3.0 },

      normal: { numBuildings: 22, baseSpeed: 1.5, maxSpeed: 4.5 },

      hard: { numBuildings: 35, baseSpeed: 2.2, maxSpeed: 6.5 }

    };


    let currentDiff = 'normal';

    let score = 0;

    let isCrashed = false;

    let keys = {};

    let safeTimeRemaining = 10;

    let safeTimerInterval = null;


    // --- DOM Elements ---

    const container = document.getElementById('canvas-container');

    const scoreEl = document.getElementById('score-val');

    const speedEl = document.getElementById('speed-val');

    const altEl = document.getElementById('alt-val');

    const crashScreen = document.getElementById('crash-screen');

    const crashReasonEl = document.getElementById('crash-reason');

    const diffSelect = document.getElementById('diff-select');

    const statusBanner = document.getElementById('status-banner');

    const countdownText = document.getElementById('countdown-text');

    const safeSubtext = document.getElementById('safe-subtext');


    // --- 1. Three.js Core Setup ---

    const scene = new THREE.Scene();

    scene.background = new THREE.Color(0x0a0a16);

    scene.fog = new THREE.FogExp2(0x0a0a16, 0.0012);


    const camera = new THREE.PerspectiveCamera(65, window.innerWidth / window.innerHeight, 0.1, 4000);

    const renderer = new THREE.WebGLRenderer({ antialias: true });

    renderer.setSize(window.innerWidth, window.innerHeight);

    renderer.setPixelRatio(Math.min(window.devicePixelRatio, 2));

    container.appendChild(renderer.domElement);


    // --- 2. Lighting ---

    scene.add(new THREE.AmbientLight(0x223344, 0.8));

    const dirLight = new THREE.DirectionalLight(0x00f0ff, 1.2);

    dirLight.position.set(100, 400, 200);

    scene.add(dirLight);


    // --- 3. Web Audio Explosions ---

    const audioCtx = new (window.AudioContext || window.webkitAudioContext)();

    function playCrashSound() {

      if (audioCtx.state === 'suspended') audioCtx.resume();

      const osc = audioCtx.createOscillator();

      const gain = audioCtx.createGain();

      osc.type = 'sawtooth';

      osc.frequency.setValueAtTime(150, audioCtx.currentTime);

      osc.frequency.exponentialRampToValueAtTime(0.01, audioCtx.currentTime + 0.8);

      gain.gain.setValueAtTime(1, audioCtx.currentTime);

      gain.gain.exponentialRampToValueAtTime(0.01, audioCtx.currentTime + 0.8);

      osc.connect(gain);

      gain.connect(audioCtx.destination);

      osc.start();

      osc.stop(audioCtx.currentTime + 0.8);

    }


    // --- 4. Jet Entity Construction ---

    const jet = new THREE.Group();

    const hullMat = new THREE.MeshStandardMaterial({ color: 0x00f0ff, roughness: 0.3, metalness: 0.8 });

    const wingMat = new THREE.MeshStandardMaterial({ color: 0xff0055, roughness: 0.4, metalness: 0.6 });


    const body = new THREE.Mesh(new THREE.ConeGeometry(2.2, 12, 6), hullMat);

    body.rotation.x = Math.PI / 2;

    jet.add(body);


    const wings = new THREE.Mesh(new THREE.BoxGeometry(18, 0.3, 3), wingMat);

    wings.position.z = 1.5;

    jet.add(wings);


    const propGeo = new THREE.BoxGeometry(0.2, 5, 0.3);

    const propMat = new THREE.MeshBasicMaterial({ color: 0x00f0ff });

    const propeller = new THREE.Mesh(propGeo, propMat);

    propeller.position.z = -6.1;

    jet.add(propeller);


    scene.add(jet);


    const planeBox = new THREE.Box3();

    const flight = {

      speed: 1.5,

      maxSpeed: DIFFICULTY[currentDiff].maxSpeed,

      minSpeed: 0.6,

      accel: 0.025,

      pitch: 0.028,

      roll: 0.038,

      yaw: 0.018

    };


    // --- 5. Dynamic World Generation with 10s Safe Zone ---

    const chunkSize = 600;

    let chunks = new Map();

    let activeObstacles = [];

    let particles = [];


    function createChunk(cx, cz) {

      const chunkGroup = new THREE.Group();

      const startX = cx * chunkSize;

      const startZ = cz * chunkSize;


      const groundMat = new THREE.MeshBasicMaterial({ color: 0x110022, wireframe: true });

      const ground = new THREE.Mesh(new THREE.PlaneGeometry(chunkSize, chunkSize, 20, 20), groundMat);

      ground.rotation.x = -Math.PI / 2;

      ground.position.set(startX + chunkSize / 2, 0, startZ + chunkSize / 2);

      chunkGroup.add(ground);


      const ceilingMat = new THREE.MeshBasicMaterial({ color: 0xff0055, wireframe: true, transparent: true, opacity: 0.15 });

      const ceiling = new THREE.Mesh(new THREE.PlaneGeometry(chunkSize, chunkSize, 10, 10), ceilingMat);

      ceiling.rotation.x = Math.PI / 2;

      ceiling.position.set(startX + chunkSize / 2, MAX_ALTITUDE, startZ + chunkSize / 2);

      chunkGroup.add(ceiling);


      // Safe Corridor: Don't spawn buildings in the initial starting area (Z > -2500)

      if (startZ < -2500) {

        const bldgMat = new THREE.MeshPhongMaterial({ color: 0x0f172a, emissive: 0x0284c7, emissiveIntensity: 0.15, flatShading: true });

        const numBldgs = DIFFICULTY[currentDiff].numBuildings;


        for (let i = 0; i < numBldgs; i++) {

          const w = 35 + Math.random() * 45;

          const h = 480 + Math.random() * 180;

          const d = 35 + Math.random() * 45;

          const bx = startX + 40 + Math.random() * (chunkSize - 80);

          const bz = startZ + 40 + Math.random() * (chunkSize - 80);


          const building = new THREE.Mesh(new THREE.BoxGeometry(w, h, d), bldgMat);

          building.position.set(bx, h / 2, bz);

          chunkGroup.add(building);


          const box = new THREE.Box3();

          box.setFromCenterAndSize(new THREE.Vector3(bx, h / 2, bz), new THREE.Vector3(w, h, d));

          activeObstacles.push({ box, passed: false, centerZ: bz });

        }

      }


      scene.add(chunkGroup);

      return chunkGroup;

    }


    function clearWorld() {

      chunks.forEach(g => scene.remove(g));

      chunks.clear();

      activeObstacles = [];

    }


    function updateChunks(px, pz) {

      const ccx = Math.floor(px / chunkSize);

      const ccz = Math.floor(pz / chunkSize);

      const keepKeys = new Set();


      for (let x = ccx - 2; x <= ccx + 2; x++) {

        for (let z = ccz - 2; z <= ccz + 2; z++) {

          const key = `${x}_${z}`;

          keepKeys.add(key);

          if (!chunks.has(key)) chunks.set(key, createChunk(x, z));

        }

      }


      chunks.forEach((group, key) => {

        if (!keepKeys.has(key)) {

          scene.remove(group);

          chunks.delete(key);

        }

      });

    }


    // --- 6. Takeoff Countdown Controller ---

    function startTakeoffSequence() {

      if (safeTimerInterval) clearInterval(safeTimerInterval);

      safeTimeRemaining = 10;

      statusBanner.style.display = 'block';

      countdownText.textContent = 'TAKEOFF ZONE';

      countdownText.style.color = '#00f0ff';

      safeSubtext.textContent = `OBSTACLES ENGAGING IN ${safeTimeRemaining}s`;


      safeTimerInterval = setInterval(() => {

        safeTimeRemaining--;

        if (safeTimeRemaining > 0) {

          safeSubtext.textContent = `OBSTACLES ENGAGING IN ${safeTimeRemaining}s`;

        } else {

          clearInterval(safeTimerInterval);

          countdownText.textContent = 'HAZARDS ACTIVE';

          countdownText.style.color = '#ff0055';

          safeSubtext.textContent = 'NAVIGATE CITY GRID';

          setTimeout(() => {

            statusBanner.style.display = 'none';

          }, 2000);

        }

      }, 1000);

    }


    // --- 7. Crash & Reset Pipeline ---

    function triggerCrash(reason) {

      if (isCrashed) return;

      isCrashed = true;

      if (safeTimerInterval) clearInterval(safeTimerInterval);

      statusBanner.style.display = 'none';


      crashReasonEl.textContent = reason;

      crashScreen.style.display = 'block';

      playCrashSound();


      const particleGeo = new THREE.BoxGeometry(1.2, 1.2, 1.2);

      const particleMat = new THREE.MeshBasicMaterial({ color: 0xff0055 });


      for (let i = 0; i < 40; i++) {

        const p = new THREE.Mesh(particleGeo, particleMat);

        p.position.copy(jet.position);

        p.velocity = new THREE.Vector3((Math.random() - 0.5) * 6, (Math.random() - 0.5) * 6, (Math.random() - 0.5) * 6);

        scene.add(p);

        particles.push(p);

      }


      jet.visible = false;

      setTimeout(resetGame, 2000);

    }


    function resetGame() {

      clearWorld();

      const settings = DIFFICULTY[currentDiff];

      flight.speed = settings.baseSpeed;

      flight.maxSpeed = settings.maxSpeed;


      jet.position.set(0, 50, 0);

      jet.quaternion.set(0, 0, 0, 1);

      

      score = 0;

      scoreEl.textContent = '0';

      crashScreen.style.display = 'none';

      jet.visible = true;

      isCrashed = false;


      particles.forEach(p => scene.remove(p));

      particles = [];


      startTakeoffSequence();

    }


    // --- 8. Event Listeners ---

    window.addEventListener('keydown', e => keys[e.code] = true);

    window.addEventListener('keyup', e => keys[e.code] = false);


    diffSelect.addEventListener('change', e => {

      currentDiff = e.target.value;

      resetGame();

    });


    document.querySelectorAll('.ctrl-btn').forEach(btn => {

      const code = btn.dataset.key;

      const setKey = val => e => { e.preventDefault(); keys[code] = val; };

      btn.addEventListener('mousedown', setKey(true));

      btn.addEventListener('mouseup', setKey(false));

      btn.addEventListener('touchstart', setKey(true));

      btn.addEventListener('touchend', setKey(false));

    });


    window.addEventListener('resize', () => {

      camera.aspect = window.innerWidth / window.innerHeight;

      camera.updateProjectionMatrix();

      renderer.setSize(window.innerWidth, window.innerHeight);

    });


    resetGame();


    // --- 9. Animation & Render Loop ---

    function animate() {

      requestAnimationFrame(animate);


      propeller.rotation.z += 0.4;


      particles.forEach(p => {

        p.position.add(p.velocity);

        p.scale.multiplyScalar(0.95);

      });


      if (!isCrashed) {

        if (keys['ShiftLeft']) flight.speed = Math.min(flight.maxSpeed, flight.speed + flight.accel);

        if (keys['ControlLeft']) flight.speed = Math.max(flight.minSpeed, flight.speed - flight.accel);


        if (keys['KeyW']) jet.rotateX(-flight.pitch);

        if (keys['KeyS']) jet.rotateX(flight.pitch);

        if (keys['KeyA']) jet.rotateZ(flight.roll);

        if (keys['KeyD']) jet.rotateZ(-flight.roll);

        if (keys['KeyQ']) jet.rotateY(flight.yaw);

        if (keys['KeyE']) jet.rotateY(-flight.yaw);


        if (!keys['KeyA'] && !keys['KeyD']) jet.rotation.z *= 0.97;


        const forward = new THREE.Vector3(0, 0, -1).applyQuaternion(jet.quaternion);

        jet.position.addScaledVector(forward, flight.speed * 6.5);


        planeBox.setFromObject(jet);


        if (jet.position.y <= 2.0) triggerCrash('Impact: Ground Surface Collision!');

        if (jet.position.y >= MAX_ALTITUDE) triggerCrash('Alert: Boundary Limits Exceeded!');


        for (let i = activeObstacles.length - 1; i >= 0; i--) {

          const obs = activeObstacles[i];

          if (planeBox.intersectsBox(obs.box)) {

            triggerCrash('Impact: Structure Collision!');

            break;

          }

          if (!obs.passed && jet.position.z < obs.centerZ - 35) {

            obs.passed = true;

            score++;

            scoreEl.textContent = score;

          }

        }


        updateChunks(jet.position.x, jet.position.z);


        speedEl.textContent = Math.round(flight.speed * 120);

        altEl.textContent = Math.round(jet.position.y);

        altEl.style.color = jet.position.y > 400 ? '#ff0055' : '#00f0ff';

      }


      // Smooth Camera Follow

      const offset = new THREE.Vector3(0, 7, 24).applyQuaternion(jet.quaternion).add(jet.position);

      const target = new THREE.Vector3(0, 0, -12).applyQuaternion(jet.quaternion).add(jet.position);

      camera.position.copy(offset);

      camera.lookAt(target);


      renderer.render(scene, camera);

    }


    animate();

  </script>
